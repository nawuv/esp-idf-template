set(portable_sources "src/sht3x.cpp")
if(COMMAND idf_component_register)
    idf_component_register(SRCS ${portable_sources}  INCLUDE_DIRS "include" REQUIRES application)
else()
    add_library(drivers STATIC ${portable_sources})
    target_include_directories(drivers PUBLIC include)
    target_link_libraries(drivers PUBLIC application)
endif()
if(COMMAND idf_component_register)
    set(component_target ${COMPONENT_LIB})
else()
    set(component_target drivers)
endif()
target_compile_features(${component_target} PUBLIC cxx_std_20)
set_target_properties(${component_target} PROPERTIES CXX_EXTENSIONS NO)
